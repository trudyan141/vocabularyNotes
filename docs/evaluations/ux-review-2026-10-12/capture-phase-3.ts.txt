import { test, expect, type Page, type TestInfo } from '@playwright/test';
import { readFile, writeFile } from 'node:fs/promises';
test.use({serviceWorkers:'block'});
test.beforeEach(async({page})=>{
 await page.addInitScript(()=>Object.defineProperty(navigator,'mediaDevices',{value:{getUserMedia:async()=>{throw new DOMException('denied','NotAllowedError');}}}));
 await page.route('https://**',r=>r.abort());
 await page.route('**/api/auth/status*',r=>r.fulfill({json:{configured:false,user:null}}));
 await page.route('**/api/illustrations?**',r=>r.fulfill({json:{result:null}}));
 await page.route('**/api/media/search**',r=>r.fulfill({json:{candidates:[],sources:[]}}));
 await page.route('**/api/dictionary**',r=>r.fulfill({status:503,json:{error:'fixture unavailable'}}));
});
async function capture(page:Page,info:TestInfo,name:string){
 await page.locator('main').waitFor();
 await page.evaluate(()=>{(document.activeElement as HTMLElement)?.blur();window.scrollTo(0,0);});
 await page.evaluate(async()=>{await document.fonts.ready;});
 await page.screenshot({path:info.outputPath(name+'.png'),fullPage:true});
 const data=await page.evaluate(()=>({url:location.pathname+location.search,title:document.querySelector('h1')?.textContent,scrollHeight:document.documentElement.scrollHeight,viewport:[innerWidth,innerHeight],overflow:document.documentElement.scrollWidth>innerWidth,text:document.querySelector('main')?.innerText??document.body.innerText,controls:[...document.querySelectorAll('main button, main a, main input, main select, main textarea, main summary')].filter(e=>e.getClientRects().length).map(e=>({text:e.textContent?.trim()||e.getAttribute('aria-label')||e.id,tag:e.tagName,w:Math.round(e.getBoundingClientRect().width),h:Math.round(e.getBoundingClientRect().height),top:Math.round(e.getBoundingClientRect().top+scrollY)}))}));
 await writeFile(info.outputPath(name+'.json'),JSON.stringify(data,null,2));
}
